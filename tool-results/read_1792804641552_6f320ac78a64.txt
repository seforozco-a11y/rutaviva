   138→async function readError(res: Response, fallback: string): Promise<string> {
   139→  try {
   140→    const j = (await res.json()) as { error?: string } | null;
   141→    return j?.error ?? fallback;
   142→  } catch {
   143→    return fallback;
   144→  }

   147→function relativeDate(iso: string): string {
   148→  try {
   149→    return formatDistanceToNow(new Date(iso), { addSuffix: true, locale: esLocale });
   150→  } catch {
   151→    return "";
   152→  }
   153→}

   155→/* ==================== Panel (puerta de sesión) ==================== */
   156→
   157→export function AdminPanel({ origin = null }: { origin?: Coordinates | null }) {
   158→  const sessionQuery = useQuery({
   159→    queryKey: ["admin-session"],
   160→    queryFn: async (): Promise<{ authenticated: boolean }> => {
   161→      const res = await fetch("/api/admin/session");
   162→      if (!res.ok) throw new Error("No se pudo verificar la sesión");
   163→      return (await res.json()) as { authenticated: boolean };
   164→    },
   165→    staleTime: 60 * 1000,
   166→  });
   167→
   168→  if (sessionQuery.isPending) {
   169→    return (
   170→      <div className="mx-auto w-full max-w-md space-y-4 px-4 pt-8" aria-busy>
   171→        <Skeleton className="mx-auto h-40 w-full rounded-2xl" />
   172→        <Skeleton className="h-11 w-full rounded-xl" />
   173→        <Skeleton className="h-24 w-full rounded-2xl" />
   174→      </div>
   175→    );
   176→  }
   177→
   178→  if (sessionQuery.isError || !sessionQuery.data) {
   179→    return (
   180→      <div className="mx-auto w-full max-w-md px-4 pt-10">
   181→        <div className="rounded-2xl border border-amber-200 bg-amber-50 p-4 text-center">
   182→          <p className="text-sm font-medium text-amber-800">
   183→            No pudimos verificar tu acceso. Revisa tu conexión e intenta de nuevo.
   184→          </p>
   185→          <Button
   186→            variant="outline"
   187→            onClick={() => void sessionQuery.refetch()}
   188→            className="mt-3 h-11 border-amber-300"
   189→          >
   190→            Reintentar
   191→          </Button>
   192→        </div>
   193→      </div>
   194→    );
   195→  }
   196→
   197→  if (!sessionQuery.data.authenticated) {
   198→    return <AdminLogin />;
   199→  }
   200→
   201→  return <AdminDashboard origin={origin} />;
   202→}

   204→/* ==================== Login con PIN ==================== */
   205→
   206→function AdminLogin() {
   207→  const { toast } = useToast();
   208→  const queryClient = useQueryClient();
   209→  const [pin, setPin] = useState("");
   210→  const [loading, setLoading] = useState(false);
   211→  const [error, setError] = useState(false);
   212→
   213→  const submit = async (e: React.FormEvent) => {
   214→    e.preventDefault();
   215→    setError(false);
   216→    setLoading(true);
   217→    try {
   218→      const res = await fetch("/api/admin/login", {
   219→        method: "POST",
   220→        headers: JSON_HEADERS,
   221→        body: JSON.stringify({ pin }),
   222→      });
   223→      if (!res.ok) {
   224→        setError(true);
   225→        return;
   226→      }
   227→      toast({
   228→        title: "¡Hola, equipo!",
   229→        description: "Bienvenidos al panel de RutaViva.",
   230→      });
   231→      queryClient.setQueryData(["admin-session"], { authenticated: true });
   232→      void queryClient.invalidateQueries({ queryKey: ["admin-session"] });
   233→    } catch {
   234→      setError(true);
   235→    } finally {
   236→      setLoading(false);
   237→    }
   238→  };
   239→
   240→  return (
   241→    <div className="mx-auto flex w-full max-w-md flex-col items-center px-4 pb-24 pt-14">
   242→      <form
   243→        onSubmit={submit}
   244→        className="w-full rounded-2xl border border-stone-200 bg-white p-6 shadow-sm"
   245→      >
   246→        <div className="flex flex-col items-center text-center">
   247→          <span className="flex h-16 w-16 items-center justify-center rounded-full bg-emerald-100 text-emerald-700">
   248→            <ShieldCheck className="h-8 w-8" aria-hidden />
   249→          </span>
   250→          <h1 className="mt-3 text-xl font-black tracking-tight text-stone-800">
   251→            Panel del equipo RutaViva
   252→          </h1>
   253→          <p className="mt-1.5 text-sm leading-relaxed text-stone-500">
   254→            Acceso exclusivo para administradores: autoriza lugares, valida reseñas y
   255→            fotografías.
   256→          </p>
   257→        </div>
   258→
   259→        <div className="mt-5 space-y-1.5">
   260→          <Label htmlFor="rv-admin-pin">PIN de acceso</Label>
   261→          <Input
   262→            id="rv-admin-pin"
   263→            type="password"
   264→            autoComplete="current-password"
   265→            value={pin}
   266→            onChange={(e) => setPin(e.target.value)}
   267→            placeholder="••••••••"
   268→            className="h-11 rounded-xl"
   269→            required
   270→          />
   271→        </div>
   272→
   273→        {error && (
   274→          <p
   275→            role="alert"
   276→            className="mt-3 animate-pulse rounded-xl bg-red-50 px-3 py-2 text-sm font-medium text-red-700"
   277→          >
   278→            PIN incorrecto
   279→          </p>
   280→        )}
   281→
   282→        <Button
   283→          type="submit"
   284→          disabled={loading}
   285→          className="mt-4 h-11 w-full gap-2 rounded-xl bg-emerald-700 font-bold hover:bg-emerald-800"
   286→        >
   287→          {loading ? (
   288→            <>
   289→              <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   290→              Verificando…
   291→            </>
   292→          ) : (
   293→            "Entrar"
   294→          )}
   295→        </Button>
   296→      </form>
   297→    </div>
   298→  );
   299→}

   301→/* ==================== Dashboard ==================== */
   302→
   303→function AdminDashboard({ origin }: { origin: Coordinates | null }) {
   304→  const { toast } = useToast();
   305→  const queryClient = useQueryClient();
   306→  const [tab, setTab] = useState("review");
   307→
   308→  const overviewQuery = useQuery({
   309→    queryKey: ["admin-overview"],
   310→    queryFn: async (): Promise<AdminOverview> => {
   311→      const res = await fetch("/api/admin/overview");
   312→      if (!res.ok) throw new Error("No se pudo cargar el panel");
   313→      return (await res.json()) as AdminOverview;
   314→    },
   315→    refetchOnWindowFocus: false,
   316→  });
   317→
   318→  const refresh = useCallback(() => {
   319→    void queryClient.invalidateQueries({ queryKey: ["admin-overview"] });
   320→  }, [queryClient]);
   321→
   322→  const refreshPublic = useCallback(() => {
   323→    void queryClient.invalidateQueries({ queryKey: ["places"] });
   324→  }, [queryClient]);
   325→
   326→  const logout = useCallback(async () => {
   327→    try {
   328→      await fetch("/api/admin/logout", { method: "POST" });
   329→    } catch {
   330→      /* la sesión caduca sola aunque falle la llamada */
   331→    }
   332→    queryClient.setQueryData(["admin-session"], { authenticated: false });
   333→    queryClient.removeQueries({ queryKey: ["admin-overview"] });
   334→    toast({ title: "Sesión cerrada", description: "Vuelve pronto." });
   335→  }, [queryClient, toast]);
   336→
   337→  /* ---- Acciones del equipo (lanzan error; la UI muestra el mensaje) ---- */
   338→
   339→  const publishValidated = useCallback(
   340→    async (pendingId: string, values: PlaceFormValues) => {
   341→      const { lat, lng } = parseCoords(values);
   342→      const updateRes = await fetch(`/api/admin/places/${pendingId}`, {
   343→        method: "PATCH",
   344→        headers: JSON_HEADERS,
   345→        body: JSON.stringify({
   346→          action: "update",
   347→          name: values.name.trim(),
   348→          description: values.description.trim(),
   349→          category: values.category,
   350→          lat,
   351→          lng,
   352→          address: values.address.trim() || null,
   353→          imageUrl: values.photos[0] ?? null,
   354→          tips: values.tips.trim() || null,
   355→          city: values.city.trim() || "Manzanillo",
   356→          state: values.state.trim() || "Colima",
   357→        }),
   358→      });
   359→      if (!updateRes.ok) {
   360→        throw new Error(await readError(updateRes, "No se pudo actualizar el lugar"));
   361→      }
   362→      const approveRes = await fetch(`/api/admin/places/${pendingId}`, {
   363→        method: "PATCH",
   364→        headers: JSON_HEADERS,
   365→        body: JSON.stringify({ action: "approve" }),
   366→      });
   367→      if (!approveRes.ok) {
   368→        throw new Error(await readError(approveRes, "No se pudo aprobar el lugar"));
   369→      }
   370→      toast({
   371→        title: "Lugar publicado",
   372→        description: `${values.name.trim()} ya aparece en el mapa para todos.`,
   373→      });
   374→      refresh();
   375→      refreshPublic();
   376→      void queryClient.invalidateQueries({ queryKey: ["place-detail", pendingId] });
   377→    },
   378→    [toast, refresh, refreshPublic, queryClient]
   379→  );
   380→
   381→  const editPlace = useCallback(
   382→    async (row: AdminPlaceRow, values: PlaceFormValues) => {
   383→      const { lat, lng } = parseCoords(values);
   384→      const res = await fetch(`/api/admin/places/${row.id}`, {
   385→        method: "PATCH",
   386→        headers: JSON_HEADERS,
   387→        body: JSON.stringify({
   388→          action: "update",
   389→          name: values.name.trim(),
   390→          description: values.description.trim(),
   391→          category: values.category,
   392→          lat,
   393→          lng,
   394→          address: values.address.trim() || null,
   395→          imageUrl: values.photos[0] ?? null,
   396→          tips: values.tips.trim() || null,
   397→          city: values.city.trim() || row.city,
   398→          state: values.state.trim() || row.state,
   399→          featured: row.featured,
   400→        }),
   401→      });
   402→      if (!res.ok) throw new Error(await readError(res, "No se pudo guardar el lugar"));
   403→      toast({ title: "Lugar actualizado", description: `${values.name.trim()} quedó al día.` });
   404→      refresh();
   405→      refreshPublic();
   406→      void queryClient.invalidateQueries({ queryKey: ["place-detail", row.id] });
   407→    },
   408→    [toast, refresh, refreshPublic, queryClient]
   409→  );
   410→
   411→  const rejectPlace = useCallback(
   412→    async (pendingId: string) => {
   413→      const res = await fetch(`/api/admin/places/${pendingId}`, {
   414→        method: "PATCH",
   415→        headers: JSON_HEADERS,
   416→        body: JSON.stringify({ action: "reject" }),
   417→      });
   418→      if (!res.ok) throw new Error(await readError(res, "No se pudo rechazar la sugerencia"));
   419→      toast({
   420→        title: "Sugerencia rechazada",
   421→        description: "No se publicará en la app.",
   422→      });
   423→      refresh();
   424→    },
   425→    [toast, refresh]
   426→  );
   427→
   428→  const reviewAction = useCallback(
   429→    async (reviewId: string, placeId: string, action: "approve" | "reject") => {
   430→      const res = await fetch(`/api/admin/reviews/${reviewId}`, {
   431→        method: "PATCH",
   432→        headers: JSON_HEADERS,
   433→        body: JSON.stringify({ action }),
   434→      });
   435→      if (!res.ok) throw new Error(await readError(res, "No se pudo actualizar la reseña"));
   436→      toast(
   437→        action === "approve"
   438→          ? { title: "Reseña publicada", description: "Ya es visible en el lugar." }
   439→          : { title: "Reseña rechazada", description: "No se publicará." }
   440→      );
   441→      refresh();
   442→      if (action === "approve") {
   443→        refreshPublic();
   444→        void queryClient.invalidateQueries({ queryKey: ["place-detail", placeId] });
   445→      }
   446→    },
   447→    [toast, refresh, refreshPublic, queryClient]
   448→  );
   449→
   450→  const featurePlace = useCallback(
   451→    async (row: AdminPlaceRow) => {
   452→      const action = row.featured ? "unfeature" : "feature";
   453→      const res = await fetch(`/api/admin/places/${row.id}`, {
   454→        method: "PATCH",
   455→        headers: JSON_HEADERS,
   456→        body: JSON.stringify({ action }),
   457→      });
   458→      if (!res.ok) throw new Error(await readError(res, "No se pudo actualizar el destaque"));
   459→      toast({
   460→        title: action === "feature" ? "Lugar destacado" : "Destaque quitado",
   461→        description: row.name,
   462→      });
   463→      refresh();
   464→      refreshPublic();
   465→    },
   466→    [toast, refresh, refreshPublic]
   467→  );
   468→
   469→  const deletePlace = useCallback(
   470→    async (row: AdminPlaceRow) => {
   471→      const res = await fetch(`/api/admin/places/${row.id}`, { method: "DELETE" });
   472→      if (!res.ok) throw new Error(await readError(res, "No se pudo eliminar el lugar"));
   473→      toast({ title: "Lugar eliminado", description: `${row.name} se borró definitivamente.` });
   474→      refresh();
   475→      refreshPublic();
   476→    },
   477→    [toast, refresh, refreshPublic]
   478→  );
   479→
   480→  const pendingTotal =
   481→    (overviewQuery.data?.stats.pendingPlaces ?? 0) +
   482→    (overviewQuery.data?.stats.pendingReviews ?? 0);
   483→
   484→  return (
   485→    <div className="mx-auto w-full max-w-md space-y-4 px-4 pb-24 pt-5">
   486→      <header className="flex items-start justify-between gap-2">
   487→        <div>
   488→          <h1 className="text-2xl font-black tracking-tight text-stone-800">Equipo RutaViva</h1>
   489→          <p className="text-sm text-stone-500">
   490→            Previo estudio se da la validación: nada se publica sin revisión.
   491→          </p>
   492→        </div>
   493→        <Button
   494→          variant="outline"
   495→          size="sm"
   496→          onClick={() => void logout()}
   497→          className="h-11 shrink-0 gap-1.5 border-stone-300 text-stone-600 hover:bg-stone-50"
   498→        >
   499→          <LogOut className="h-4 w-4" aria-hidden />
   500→          Salir
   501→        </Button>
   502→      </header>
   503→
   504→      {overviewQuery.isPending ? (
   505→        <div className="space-y-4" aria-busy>
   506→          <div className="grid grid-cols-3 gap-2">
   507→            <Skeleton className="h-20 rounded-2xl" />
   508→            <Skeleton className="h-20 rounded-2xl" />
   509→            <Skeleton className="h-20 rounded-2xl" />
   510→          </div>
   511→          <Skeleton className="h-11 w-full rounded-xl" />
   512→          <Skeleton className="h-48 w-full rounded-2xl" />
   513→        </div>
   514→      ) : overviewQuery.isError || !overviewQuery.data ? (
   515→        <div className="rounded-2xl border border-amber-200 bg-amber-50 p-4 text-center">
   516→          <p className="text-sm font-medium text-amber-800">
   517→            No pudimos cargar la información del panel.
   518→          </p>
   519→          <Button
   520→            variant="outline"
   521→            onClick={() => void overviewQuery.refetch()}
   522→            className="mt-3 h-11 border-amber-300"
   523→          >
   524→            Reintentar
   525→          </Button>
   526→        </div>
   527→      ) : (
   528→        <>
   529→          <section
   530→            aria-label="Resumen del panel"
   531→            className="grid grid-cols-3 gap-2"
   532→          >
   533→            <StatCard
   534→              value={overviewQuery.data.stats.placesPublished}
   535→              label="Publicados"
   536→              alert={false}
   537→            />
   538→            <StatCard
   539→              value={overviewQuery.data.stats.pendingPlaces}
   540→              label="Sugerencias por revisar"
   541→              alert={overviewQuery.data.stats.pendingPlaces > 0}
   542→            />
   543→            <StatCard
   544→              value={overviewQuery.data.stats.pendingReviews}
   545→              label="Reseñas por validar"
   546→              alert={overviewQuery.data.stats.pendingReviews > 0}
   547→            />
   548→          </section>
   549→
   550→          <Tabs value={tab} onValueChange={setTab}>
   551→            <TabsList className="grid h-auto w-full grid-cols-3 gap-1 rounded-2xl bg-stone-100 p-1">
   552→              <TabsTrigger
   553→                value="review"
   554→                className="min-h-[40px] gap-1 rounded-xl text-xs font-bold data-[state=active]:bg-white"
   555→              >
   556→                Por revisar
   557→                {pendingTotal > 0 && (
   558→                  <span className="inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-amber-400 px-1 text-[10px] font-black text-amber-950">
   559→                    {pendingTotal}
   560→                  </span>
   561→                )}
   562→              </TabsTrigger>
   563→              <TabsTrigger
   564→                value="places"
   565→                className="min-h-[40px] rounded-xl text-xs font-bold data-[state=active]:bg-white"
   566→              >
   567→                Lugares
   568→              </TabsTrigger>
   569→              <TabsTrigger
   570→                value="new"
   571→                className="min-h-[40px] rounded-xl text-xs font-bold data-[state=active]:bg-white"
   572→              >
   573→                Nuevo lugar
   574→              </TabsTrigger>
   575→            </TabsList>
   576→
   577→            <TabsContent value="review" className="mt-4">
   578→              <ReviewQueue
   579→                pendingPlaces={overviewQuery.data.pendingPlaces}
   580→                pendingReviews={overviewQuery.data.pendingReviews}
   581→                origin={origin}
   582→                onPublishValidated={publishValidated}
   583→                onRejectPlace={rejectPlace}
   584→                onReviewAction={reviewAction}
   585→              />
   586→            </TabsContent>
   587→
   588→            <TabsContent value="places" className="mt-4">
   589→              <PlacesSection
   590→                places={overviewQuery.data.allPlaces}
   591→                onEditSave={editPlace}
   592→                onFeature={featurePlace}
   593→                onDelete={deletePlace}
   594→              />
   595→            </TabsContent>
   596→
   597→            <TabsContent value="new" className="mt-4">
   598→              <NewPlaceSection origin={origin} onCreated={refresh} onPublished={refreshPublic} />
   599→            </TabsContent>
   600→          </Tabs>
   601→        </>
   602→      )}
   603→    </div>

   914→function PlaceFormDialogBody(props: {
   915→  idPrefix: string;
   916→  title: string;
   917→  description: string;
   918→  initial: PlaceFormValues;
   919→  origin: Coordinates | null;
   920→  requireChecks: boolean;
   921→  saveLabel: string;
   922→  onSave: (values: PlaceFormValues) => Promise<void>;
   923→  onClose: () => void;
   924→}) {
   925→  const [values, setValues] = useState<PlaceFormValues>(props.initial);
   926→  const [saving, setSaving] = useState(false);
   927→  const [error, setError] = useState<string | null>(null);
   928→  const checks = placeChecks(values);
   929→  const complete = checksComplete(checks);
   930→
   931→  const patch = (p: Partial<PlaceFormValues>) => setValues((v) => ({ ...v, ...p }));
   932→
   933→  const save = async (e: React.FormEvent) => {
   934→    e.preventDefault();
   935→    setError(null);
   936→    if (values.name.trim().length < 3)
   937→      return setError("El nombre debe tener al menos 3 caracteres.");
   938→    if (values.description.trim().length < 10)
   939→      return setError("La descripción debe tener al menos 10 caracteres.");
   940→    const { lat, lng } = parseCoords(values);
   941→    if (!Number.isFinite(lat) || !Number.isFinite(lng))
   942→      return setError("Captura una latitud y longitud válidas.");
   943→
   944→    setSaving(true);
   945→    try {
   946→      await props.onSave(values);
   947→      props.onClose();
   948→    } catch (err) {
   949→      setError(err instanceof Error ? err.message : "Error inesperado");
   950→    } finally {
   951→      setSaving(false);
   952→    }
   953→  };
   954→
   955→  return (
   956→    <form onSubmit={save} className="space-y-4">
   957→      <DialogHeader className="text-left">
   958→        <DialogTitle className="text-lg font-black text-stone-800">{props.title}</DialogTitle>
   959→        <DialogDescription className="text-sm leading-relaxed text-stone-500">
   960→          {props.description}
   961→        </DialogDescription>
   962→      </DialogHeader>
   963→
   964→      <PlaceFieldsForm
   965→        idPrefix={props.idPrefix}
   966→        values={values}
   967→        onPatch={patch}
   968→        origin={props.origin}
   969→        showChecklist={props.requireChecks}
   970→      />
   971→
   972→      {error && (
   973→        <p role="alert" className="rounded-xl bg-red-50 px-3 py-2 text-sm font-medium text-red-700">
   974→          {error}
   975→        </p>
   976→      )}
   977→
   978→      <DialogFooter className="flex-col gap-2">
   979→        {props.requireChecks && !complete && (
   980→          <p className="w-full text-center text-xs font-medium text-amber-700">
   981→            Completa los formatos para publicar
   982→          </p>
   983→        )}
   984→        <div className="flex w-full gap-2">
   985→          <Button
   986→            type="submit"
   987→            disabled={saving || (props.requireChecks && !complete)}
   988→            className="h-11 flex-1 gap-2 rounded-xl bg-emerald-700 font-bold hover:bg-emerald-800"
   989→          >
   990→            {saving ? (
   991→              <>
   992→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   993→                Guardando…
   994→              </>
   995→            ) : (
   996→              props.saveLabel
   997→            )}
   998→          </Button>
   999→          <Button
  1000→            type="button"
  1001→            variant="outline"
  1002→            onClick={props.onClose}
  1003→            disabled={saving}
  1004→            className="h-11 rounded-xl border-stone-300"
  1005→          >
  1006→            Cancelar
  1007→          </Button>
  1008→        </div>
  1009→      </DialogFooter>
  1010→    </form>
  1011→  );
  1012→}

  1233→function NewPlaceSection(props: {
  1234→  origin: Coordinates | null;
  1235→  onCreated: () => void;
  1236→  onPublished: () => void;
  1237→}) {
  1238→  const { toast } = useToast();
  1239→  const [values, setValues] = useState<PlaceFormValues>(emptyPlaceValues());
  1240→  const [featured, setFeatured] = useState(false);
  1241→  const [saving, setSaving] = useState(false);
  1242→  const [error, setError] = useState<string | null>(null);
  1243→
  1244→  const patch = (p: Partial<PlaceFormValues>) => setValues((v) => ({ ...v, ...p }));
  1245→
  1246→  const submit = async (e: React.FormEvent) => {
  1247→    e.preventDefault();
  1248→    setError(null);
  1249→    if (values.name.trim().length < 3)
  1250→      return setError("El nombre debe tener al menos 3 caracteres.");
  1251→    if (values.description.trim().length < 10)
  1252→      return setError("La descripción debe tener al menos 10 caracteres.");
  1253→    const { lat, lng } = parseCoords(values);
  1254→    if (!Number.isFinite(lat) || !Number.isFinite(lng))
  1255→      return setError("Captura una latitud y longitud válidas (o usa tu ubicación).");
  1256→
  1257→    setSaving(true);
  1258→    try {
  1259→      const res = await fetch("/api/admin/places", {
  1260→        method: "POST",
  1261→        headers: JSON_HEADERS,
  1262→        body: JSON.stringify({
  1263→          name: values.name.trim(),
  1264→          description: values.description.trim(),
  1265→          category: values.category,
  1266→          lat,
  1267→          lng,
  1268→          address: values.address.trim() || null,
  1269→          imageUrl: values.photos[0] ?? null,
  1270→          tips: values.tips.trim() || null,
  1271→          city: values.city.trim() || "Manzanillo",
  1272→          state: values.state.trim() || "Colima",
  1273→          featured,
  1274→        }),
  1275→      });
  1276→      if (!res.ok) throw new Error(await readError(res, "No se pudo publicar el lugar"));
  1277→      toast({
  1278→        title: "Lugar publicado en el mapa",
  1279→        description: `${values.name.trim()} ya está disponible para todos.`,
  1280→      });
  1281→      setValues(emptyPlaceValues());
  1282→      setFeatured(false);
  1283→      props.onCreated();
  1284→      props.onPublished();
  1285→    } catch (err) {
  1286→      setError(err instanceof Error ? err.message : "Error inesperado");
  1287→    } finally {
  1288→      setSaving(false);
  1289→    }
  1290→  };
  1291→
  1292→  return (
  1293→    <form onSubmit={submit} className="space-y-4">
  1294→      <h2 className="text-sm font-black uppercase tracking-wide text-stone-500">
  1295→        Alta directa del equipo
  1296→      </h2>
  1297→      <p className="rounded-xl bg-emerald-50 px-3 py-2 text-xs leading-relaxed text-emerald-900">
  1298→        Los lugares que publicas aquí nacen aprobados y visibles de inmediato.
  1299→      </p>
  1300→
  1301→      <PlaceFieldsForm
  1302→        idPrefix="rv-new"
  1303→        values={values}
  1304→        onPatch={patch}
  1305→        origin={props.origin}
  1306→        showCityState
  1307→      />
  1308→
  1309→      <div className="flex items-center justify-between rounded-xl border border-stone-200 bg-white px-3 py-2.5">
  1310→        <Label htmlFor="rv-new-featured" className="flex items-center gap-2 text-sm">
  1311→          <Star className="h-4 w-4 fill-amber-400 text-amber-400" aria-hidden />
  1312→          Destacar lugar
  1313→        </Label>
  1314→        <Switch
  1315→          id="rv-new-featured"
  1316→          checked={featured}
  1317→          onCheckedChange={setFeatured}
  1318→          aria-label="Destacar lugar"
  1319→        />
  1320→      </div>
  1321→
  1322→      {error && (
  1323→        <p role="alert" className="rounded-xl bg-red-50 px-3 py-2 text-sm font-medium text-red-700">
  1324→          {error}
  1325→        </p>
  1326→      )}
  1327→
  1328→      <Button
  1329→        type="submit"
  1330→        disabled={saving}
  1331→        className="h-11 w-full gap-2 rounded-xl bg-emerald-700 font-bold hover:bg-emerald-800"
  1332→      >
  1333→        {saving ? (
  1334→          <>
  1335→            <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
  1336→            Publicando…
  1337→          </>
  1338→        ) : (
  1339→          "Publicar lugar"
  1340→        )}
  1341→      </Button>
  1342→    </form>
  1343→  );
  1344→}
